// 3. Inicializar Firebase y Firestore const app = initializeApp(firebaseConfig); const db = getFirestore(app); // Variables globales para el flujo local let historialDatos = []; // 4. Función encargada de enviar el registro a Firestore de forma automática async function guardarEnNube(columnas) { try { // Creamos el objeto con la estructura de 4 columnas const nuevoRegistro = { columna1: columnas[0] || '-', columna2: columnas[1] || '-', columna3: columnas[2] || '-', columna4: columnas[3] || '-', fechaServidor: serverTimestamp() // Registra la hora exacta en la nube }; // Enviamos el documento a una colección llamada "escaneos" const docRef = await addDoc(collection(db, "escaneos"), nuevoRegistro); console.log("¡Guardado en la nube con ID:!", docRef.id); // Actualizar interfaz local historialDatos.push(nuevoRegistro); actualizarTabla(); } catch (error) { console.error("Error al guardar en la nube: ", error); alert("Error de conexión. No se pudo guardar en la nube."); } } // --- Procesamiento del QR --- function procesarDatosQR(texto) { let columnas = texto.split(';'); // Separa el texto por puntos y comas guardarEnNube(columnas); // Llama a la función de la nube } function actualizarTabla() { const body = document.getElementById('historial-body'); body.innerHTML = ''; historialDatos.forEach(reg => { body.innerHTML += ` ${reg.columna1}${reg.columna2} ${reg.columna3}${reg.columna4} `; }); } // --- Configuración básica del Escáner QR --- function onScanSuccess(decodedText) { document.getElementById('result-status').style.display = "block"; document.getElementById('result-status').innerText = `QR Detectado`; procesarDatosQR(decodedText); } const html5QrcodeScanner = new Html5QrcodeScanner("reader", { fps: 10, qrbox: 250 }); html5QrcodeScanner.render(onScanSuccess); // Hacer la función de exportar accesible globalmente window.exportarCSV = function() { if(historialDatos.length === 0) return alert("No hay datos"); let csv = "data:text/csv;charset=utf-8,Col1,Col2,Col3,Col4\n"; historialDatos.forEach(r => csv += `"${r.columna1}","${r.columna2}","${r.columna3}","${r.columna4}"\n`); const link = document.createElement("a"); link.setAttribute("href", encodeURI(csv)); link.setAttribute("download", "historial.csv"); link.click(); }
